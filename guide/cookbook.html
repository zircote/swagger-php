Cookbook ​

Return JSON from an endpoint ​

For application/json content, OA\MediaType\Json takes the schema directly, without a nested OA\Schema. OA\MediaType\Xml is the same for XML.

php
<?php

namespace Openapi\Snippets\Cookbook\JsonResponse;

use OpenApi\Spec as OA;

class PetController
{
    #[OA\Operation\Get(path: '/pets/current')]
    #[OA\Response(
        response: 200,
        description: 'The current pet',
        content: [new OA\MediaType\Json(ref: '#/components/schemas/Pet')],
    )]
    public function current()
    {
    }
}
php
<?php

namespace Openapi\Snippets\Cookbook\JsonResponse;

use OpenApi\Attributes as OA;

class PetController
{
    #[OA\Get(path: '/pets/current')]
    #[OA\Response(
        response: 200,
        description: 'The current pet',
        content: new OA\JsonContent(ref: '#/components/schemas/Pet'),
    )]
    public function current()
    {
    }
}
php
<?php

namespace Openapi\Snippets\Cookbook\JsonResponse;

use OpenApi\Annotations as OA;

class PetController
{
    /**
     * @OA\Get(
     *     path="/pets/current",
     *     @OA\Response(
     *         response=200,
     *         description="The current pet",
     *         @OA\JsonContent(ref="#/components/schemas/Pet")
     *     )
     * )
     */
    public function current()
    {
    }
}

See OA\MediaType.

Document a list of objects ​

A property holding a list takes OA\Schema\Items, with the schema of one item as its arguments.

php
<?php

namespace Openapi\Snippets\Cookbook\ListOfObjects;

use OpenApi\Spec as OA;

#[OA\Schema(component: 'Pet')]
class Pet
{
    #[OA\Schema\Items(ref: '#/components/schemas/Tag')]
    public array $tags;
}
php
<?php

namespace Openapi\Snippets\Cookbook\ListOfObjects;

use OpenApi\Attributes as OA;

#[OA\Schema(schema: 'Pet')]
class Pet
{
    #[OA\Property(type: 'array', items: new OA\Items(ref: '#/components/schemas/Tag'))]
    public array $tags;
}
php
<?php

namespace Openapi\Snippets\Cookbook\ListOfObjects;

use OpenApi\Annotations as OA;

/**
 * @OA\Schema(schema="Pet")
 */
class Pet
{
    /**
     * @OA\Property(type="array", @OA\Items(ref="#/components/schemas/Tag"))
     */
    public array $tags;
}

See OA\Schema subclasses.

Describe an object without a PHP class ​

Where a controller returns an array rather than an object of a class, properties can be keyed by property name. Classic has no short form for this.

php
<?php

namespace Openapi\Snippets\Cookbook\ObjectWithoutClass;

use OpenApi\Spec as OA;

class UserController
{
    #[OA\Operation\Get(path: '/users/me')]
    #[OA\Response(
        response: 200,
        description: 'The signed in user',
        content: [new OA\MediaType\Json(properties: [
            'name' => OA\Type::String,
            'email' => OA\Format::Email,
            'lastLogin' => OA\Format::DateTime,
        ])],
    )]
    public function me()
    {
    }
}
php
<?php

namespace Openapi\Snippets\Cookbook\ObjectWithoutClass;

use OpenApi\Attributes as OA;

class UserController
{
    #[OA\Get(path: '/users/me')]
    #[OA\Response(
        response: 200,
        description: 'The signed in user',
        content: new OA\JsonContent(properties: [
            new OA\Property(property: 'name', type: 'string'),
            new OA\Property(property: 'email', type: 'string', format: 'email'),
            new OA\Property(property: 'lastLogin', type: 'string', format: 'date-time'),
        ]),
    )]
    public function me()
    {
    }
}
php
<?php

namespace Openapi\Snippets\Cookbook\ObjectWithoutClass;

use OpenApi\Annotations as OA;

class UserController
{
    /**
     * @OA\Get(
     *     path="/users/me",
     *     @OA\Response(
     *         response=200,
     *         description="The signed in user",
     *         @OA\JsonContent(
     *             @OA\Property(property="name", type="string"),
     *             @OA\Property(property="email", type="string", format="email"),
     *             @OA\Property(property="lastLogin", type="string", format="date-time")
     *         )
     *     )
     * )
     */
    public function me()
    {
    }
}

    ],

{
}
php
<?php

#[OA\Schema(
    schema: 'Profile',
    properties: [
        new OA\Property(property: 'status', type: 'string'),

            new OA\Property(property: 'id', type: 'integer'),

/**

{

Add constraints to a class property ​

A class property needs only an OA\Schema with the constraints. The name and the type come from the declaration. Spec also lists a typed property without a default as required.

php
<?php

namespace Openapi\Snippets\Cookbook\PropertyConstraints;

use OpenApi\Spec as OA;

#[OA\Schema(component: 'User')]
class User
{
    #[OA\Schema(maxLength: 50)]
    public string $name;

    #[OA\Schema(format: 'email')]
    public string $email;
}
php
<?php

namespace Openapi\Snippets\Cookbook\PropertyConstraints;

use OpenApi\Attributes as OA;

#[OA\Schema(schema: 'User')]
class User
{
    #[OA\Property(maxLength: 50)]
    public string $name;

    #[OA\Property(format: 'email')]
    public string $email;
}
php
<?php

namespace Openapi\Snippets\Cookbook\PropertyConstraints;

use OpenApi\Annotations as OA;

/**
 * @OA\Schema(schema="User")
 */
class User
{
    /**
     * @OA\Property(maxLength=50)
     */
    public string $name;

    /**
     * @OA\Property(format="email")
     */
    public string $email;
}

An explicit OA\Property is needed only where the document uses a different name, such as full_name for $name, or on a getter. See OA\Property.

Document path, query and header parameters ​

OA\Parameter\Path, Query, Header and Cookie set in, and a path parameter is always required.

php

namespace Openapi\Snippets\Cookbook\Parameters;

            new OA\Parameter\Header(name: 'X-Request-Id', schema: new OA\Schema(type: 'string')),
        ],
    )]
    #[OA\Response(response: 200, description: 'The pet')]
    public function show()
    {
    }
}

php

namespace Openapi\Snippets\Cookbook\ResponseExamples;

use OpenApi\Spec as OA;

    schema: 'Result',
    type: 'object',

{
}

Adding examples to OA\Schema ​

A schema's examples is the JSON Schema keyword rather than a map of Example Objects, so it takes a list of values. OA\Schema\Items and OA\Schema\AdditionalProperties are schemas too and take the same list. A property takes it on the OA\Schema beside its OA\Property.

php
<?php

namespace Openapi\Snippets\Cookbook\SchemaExamples;

use OpenApi\Spec as OA;

#[OA\Schema(
    schema: 'Port',
    type: 'integer',
    examples: [80, 443],
)]
class PortModel
{
}
php
<?php

namespace Openapi\Snippets\Cookbook\SchemaExamples;

use OpenApi\Attributes as OA;

#[OA\Schema(
    schema: 'Port',
    type: 'integer',
    examples: [80, 443],
)]
class PortModel
{
}
php
<?php

namespace Openapi\Snippets\Cookbook\SchemaExamples;

use OpenApi\Annotations as OA;

/**
 * @OA\Schema(
 *     schema="Port",
 *     type="integer",
 *     examples={80, 443}
 * )
 */
class PortModel
{
}

A media type, parameter or header takes Example Objects instead.

Mode difference

In classic, OA\Property, OA\Items and OA\AdditionalProperties are schemas themselves and take the list directly. A nested OA\Examples is accepted there and contributes its value. The rest of it (the key, summary, description, externalValue) is dropped. In spec, an Example Object in the list is reported with a warning.

The keyword arrived in OpenAPI 3.1. Generating 3.0 from the same source drops the list in classic and carries the first value across as example in spec. Both warn.

External documentation ​

OpenApi allows a single reference to external documentation. This is a part of the top level OA\OpenApi.

php
<?php

namespace Openapi\Snippets\Cookbook\ExternalDocumentation;

use OpenApi\Spec as OA;

#[OA\ExternalDocumentation(
    description: 'More documentation here...',
    url: 'https://example.com/externaldoc1/',
)]
class OpenApiSpec
{
}
php
<?php

namespace Openapi\Snippets\Cookbook\ExternalDocumentation;

use OpenApi\Attributes as OA;

#[OA\OpenApi(
    externalDocs: new OA\ExternalDocumentation(
        description: 'More documentation here...',
        url: 'https://example.com/externaldoc1/',
    ),
)]
class OpenApiSpec
{
}
php
<?php

namespace Openapi\Snippets\Cookbook\ExternalDocumentation;

use OpenApi\Annotations as OA;

/**
 * @OA\OpenApi(
 *     @OA\ExternalDocumentation(
 *         description="More documentation here...",
 *         url="https://example.com/externaldoc1/"
 *     )
 * )
 */
class OpenApiSpec
{
}

TIP

If no OA\OpenApi is configured, swagger-php will create one automatically.

That means the above example would also work with just the OA\ExternalDocumentation.

php
#[OA\ExternalDocumentation(
    description: 'More documentation here...',
    url: 'https://example.com/externaldoc1/',
)]

Properties with union types ​

Sometimes properties or even lists (arrays) may contain data of different types. This can be expressed using oneOf.

php
<?php

namespace Openapi\Snippets\Cookbook\UnionTypes;

use OpenApi\Spec as OA;

#[OA\Schema(
    schema: 'StringList',
    properties: [
        new OA\Property(
            property: 'value',
            schema: new OA\Schema(
                type: 'array',
                items: new OA\Schema(
                    anyOf: [new OA\Schema(type: 'string')]
                ),
            ),
        ),
    ],
)]
#[OA\Schema(
    schema: 'String',
    properties: [
        new OA\Property(
            property: 'value',
            schema: new OA\Schema(type: 'string'),
        ),
],
)]
#[OA\Schema(
    schema: 'Object',
    properties: [
        new OA\Property(
            property: 'value',
            schema: new OA\Schema(type: OA\Type::Object),
        ),
],
)]
#[OA\Schema(
    schema: 'mixedList',
    properties: [
        new OA\Property(
            property: 'fields',
            schema: new OA\Schema(
                type: 'array',
                items: new OA\Schema(
                    oneOf: [
                    new OA\Schema(ref: '#/components/schemas/StringList'),
                    new OA\Schema(ref: '#/components/schemas/String'),
                    new OA\Schema(ref: '#/components/schemas/Object'),
                ],
                ),
            ),
        ),
    ],
)]
class OpenApiSpec
{
}
php
<?php

namespace Openapi\Snippets\Cookbook\UnionTypes;

Referencing a security scheme ​

An API might have zero or more security schemes. These are defined at the top level and vary from simple to complex:

php
<?php

namespace Openapi\Snippets\Cookbook\SecuritySchemas;

use OpenApi\Spec as OA;

#[OA\Security\Scheme\ApiKey(
    name: 'api_key',
    component: 'api_key',
    in: OA\SchemeIn::Header,
)]
#[OA\Security\Scheme\OAuth2(
    component: 'petstore_auth',
    flows: [
        new OA\Flow(
            flow: 'implicit',
            authorizationUrl: 'http://petstore.swagger.io/oauth/dialog',
            scopes: [
                'read:pets' => 'read your pets',
                'write:pets' => 'modify pets in your account',
            ],
        ),
    ],
)]
class OpenApiSpec
{
}
php
<?php

namespace Openapi\Snippets\Cookbook\SecuritySchemas;

use OpenApi\Attributes as OA;

#[OA\SecurityScheme(
    type: 'apiKey',
    name: 'api_key',
    in: 'header',
    securityScheme: 'api_key',
)]
#[OA\SecurityScheme(
    type: 'oauth2',
    securityScheme: 'petstore_auth',
    flows: [
        new OA\Flow(
            authorizationUrl: 'http://petstore.swagger.io/oauth/dialog',
            flow: 'implicit',
            scopes: [
                'read:pets' => 'read your pets',
                'write:pets' => 'modify pets in your account',
            ],
        ),
    ],
)]
class OpenApiSpec
{
}
php
<?php

namespace Openapi\Snippets\Cookbook\SecuritySchemas;

use OpenApi\Annotations as OA;

/**
 * @OA\SecurityScheme(
 *     type="apiKey",
 *     name="api_key",
 *     in="header",
 *     securityScheme="api_key"
 * )
 *
 * @OA\SecurityScheme(
 *     type="oauth2",
 *     securityScheme="petstore_auth",
 *     @OA\Flow(
 *         authorizationUrl="http://petstore.swagger.io/oauth/dialog",
 *         flow="implicit",
 *         scopes={
 *             "read:pets": "read your pets",
 *             "write:pets": "modify pets in your account"
 *         }
 *     )
 * )
 */
class OpenApiSpec
{
}

To declare an endpoint as secure and define what security schemes are available to authenticate a client, it needs to be added to the operation, for example:

php
<?php

namespace Openapi\Snippets\Cookbook\SecureEndpoint;

use OpenApi\Spec as OA;

class Controller
{
    #[OA\Operation\Get(
        path: '/api/secure/',
        summary: 'Requires authentication',
        security: [
            new OA\Security\Requirement(scheme: 'api_key', scopes: []),
        ],
    )]
    public function getSecurely()
    {
        // ...
    }
}
php
<?php

namespace Openapi\Snippets\Cookbook\SecureEndpoint;

use OpenApi\Attributes as OA;

class Controller
{
    #[OA\Get(
        path: '/api/secure/',
        summary: 'Requires authentication',
        security: [
            [
                'api_key' => [],

response parameter is always required

mediaType="/" ​

Using */* as mediaType is not possible with docblock annotations. Attributes do not have this problem.

Example:

php
/**
 * @OA\MediaType(
 *     mediaType="*/*",
 *     @OA\Schema(type="string",format="binary")
 * )
 */

The doctrine annotations library used for parsing annotations does not handle this and will interpret the */ bit as the end of the comment.

Using just * or application/octet-stream might be usable workarounds.

Using class name as a type instead of references ​

Typically, when referencing schemas this is done using $ref's

php
#[OA\Schema(component: 'user')]
class User
{
}

#[OA\Schema]
class Book
{
    /**
     * @var User
     */
    #[OA\Schema(ref: '#/components/schemas/user')]
    public $author;
}

This works but is not very convenient.

First, when using custom schema names (component: 'user'), this needs to be taken into account everywhere. Secondly, having to write ref: '#/components/schemas/user' is tedious and error-prone.

Using attributes, all this changes as we can take advantage of PHP itself by referring to a schema by its (fully qualified) class name.

With the same User schema as before, the Book::author property could be written in a few different ways

php
    #[OA\Property]
    public User $author;

or

php
    /**
     * @var User
     */
    #[OA\Property]
    public $author;

or

php
    #[OA\Schema(ref: User::class)]
    public $author;
php
    #[OAT\Property()]
    public User $author;

or

php
    /**
     * @var User
     */
    #[OAT\Property()]
    public $author;

or

php
    #[OAT\Property(type: User::class)]
    public $author;
php
    /** @OA\Property() */
    public User $author;

or

php
    /**
     * @var User
     * @OA\Property()
     */
    public $author;

Custom response classes ​

Even with using refs, there is a bit of overhead in sharing responses. One way around that is to write your own response classes. The beauty is that in your custom __construct() method you can prefill as much as you need.

Example:

php
use OpenApi\Spec as OA;

#[\Attribute(\Attribute::TARGET_CLASS | \Attribute::TARGET_METHOD | \Attribute::IS_REPEATABLE)]
class BadRequest extends OA\Response
{
    public function __construct()
    {
        parent::__construct(response: 400, description: 'Bad request');
    }
}

class Controller
{
    #[OA\Operation\Get(path: '/foo', responses: [new BadRequest()])]
    public function get()
    {
    }

    #[OA\Operation\Post(path: '/foo')]
    #[BadRequest]
    public function post()
    {
    }
}

Classic attributes and docblock annotations ​

php
use OpenApi\Attributes as OA;

/**
 * @Annotation
 */
#[\Attribute(\Attribute::TARGET_CLASS | \Attribute::TARGET_METHOD | \Attribute::IS_REPEATABLE)]
class BadRequest extends OA\Response

class Controller
{

    public function get()

    }

    #[OA\Post(path: '/foo')]

     * @OA\Delete(
     *     path="/foo",

namespace Openapi\Snippets\Cookbook\ClassConstants;

 * @OA\Schema

class Airport

    /** @OA\Property(property="kind") */
    public const KIND = 'Airport';

The const property is supported in OpenApi 3.1.0.

yaml
components:
  schemas:
    Airport:
        properties:
          kind:
            type: string
            const: Airport

For 3.0.0 this is serialized into a single value enum.

yaml
components:
  schemas:
    Airport:
        properties:
          kind:
            type: string
            enum:
              - Airport

Build parts of a document for another one ​

An integration that keeps a document of its own can still take parts of it from spec attributes: a model's schema, or a parameter. Build each part with a Builder, so it goes through the whole pipeline: the resolver, the augmenters and any augmenter the integration registers. Compiling the attributes directly skips all of that.

php
<?php declare(strict_types=1);

namespace OpenApi\Snippets\Guide\Cookbook\Fragments;

use OpenApi\Augmenter\Cleanup;
use OpenApi\Builder;
use OpenApi\Builder\Mode;
use OpenApi\Contracts\AttributeInterface;
use OpenApi\Spec as OA;
use OpenApi\Specification;
use OpenApi\Utils\Pipeline;

/**
 * A new builder for each part; the host's own augmenters are registered here too.
 */
function partBuilder(): Builder
{
    return (new Builder())
        ->setMode(Mode::SPEC)
        ->setVersion('3.1.0')
        ->withAugmenters(fn (Pipeline $augmenters) => $augmenters->remove(Cleanup::class))
        ->withSpecification(fn (Specification $specification) => $specification->add(
            (new OA\Info(title: 'Part', version: '1.0.0'))->setMeta('host.filler', true)
        ));
}

/**
 * The schema of a class with `#[OA\Schema]`.
 *
 * @param class-string $class
 *
 * @return array<string, mixed>
 */
function buildModel(string $class): array
{
    $reflector = new \ReflectionClass($class);
    $document = partBuilder()->addSource($reflector)->build()->toArray();

    return $document['components']['schemas'][$reflector->getShortName()];
}

/**
 * A part keyed with `component:`, from the given `components` bucket.
 *
 * @return array<string, mixed>
 */
function buildPart(AttributeInterface $part, string $bucket): array
{
    $document = partBuilder()
        ->withSpecification(fn (Specification $specification) => $specification->add($part))
        ->build()
        ->toArray();

    return $document['components'][$bucket][$part->component];
}

The builder:

  • removes the Cleanup augmenter. Nothing references a part, so Cleanup would remove it as unused.
  • adds an OA\Info, since a document without one reports an error. It is filler, marked with setMeta() under a key the integration owns, so its own augmenters can leave it alone.
  • is new for every part. A Builder builds one document, and building a few classes is cheap.

The integration then reads each part by its key and ignores the rest of the document. For the classes below, buildModel(User::class) and a path parameter passed to buildPart():

php
<?php declare(strict_types=1);

namespace OpenApi\Snippets\Guide\Cookbook\Fragments;

use OpenApi\Spec as OA;

#[OA\Schema(description: 'A registered user')]
class User
{
    /**
     * @example 7
     */
    #[OA\Property(schema: new OA\Schema(readOnly: true))]
    public int $id;

    #[OA\Property(property: 'display_name')]
    public ?string $name = null;

    #[OA\Property]
    public Address $address;
}

#[OA\Schema]
class Address
{
    #[OA\Property]
    public string $street;
}
php
buildPart(new OA\Parameter\Path(name: 'id', schema: new OA\Schema(type: 'integer'), component: 'id'), 'parameters');

give:

yaml
User:
  type: object
  description: 'A registered user'
  properties:
    id:
      type: integer
      readOnly: true
      example: 7
    display_name:
      type:
        - string
        - 'null'
    address:
      $ref: '#/components/schemas/Address'
  required:
    - id
    - address
id:
  name: id
  in: path
  required: true
  schema:
    type: integer

A $ref in a part points into the components of the build that made it, as #/components/schemas/Address does here. The integration maps it to the name it uses for that class in its own document.

At version 3.0.0 the build also warns that paths is required. An OA\Operation added as filler next to the OA\Info removes the warning.
